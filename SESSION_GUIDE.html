2. Start of Session

Enter project

# Sister
cd /home/jose-a-pichardo/Wan2.2
# Primary (SSH)
ssh jarvis@100.105.156.94
cd /home/jarvis/Wan2.2


Check state


git status
git log --oneline -5
git branch -a

Sync with upstream


git fetch upstream
git log upstream/main --oneline -3
git merge upstream/main

Start services


# Terminal 1 - ComfyUI
cd /home/jarvis/ComfyUI
source /home/jarvis/.venv/wan2.2/bin/activate
python3 main.py --listen 0.0.0.0 --port 8188

# Terminal 2 - FastAPI wrapper
cd /home/jarvis/Wan2.2
source /home/jarvis/.venv/wan2.2/bin/activate
python3 -m uvicorn wan_service.main:app --host 0.0.0.0 --port 8765
